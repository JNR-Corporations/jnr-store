// Generated. Online requests bypass HTTP cache. Only the shell has offline fallback.
const CACHE='jnr-shell-__VERSION__';
const SHELL=__SHELL__;
const scope=self.registration.scope;
const canonical=value=>{const u=new URL(value,scope);u.search='';u.hash='';return u.href;};
const allowed=new Set(SHELL.map(canonical));
const entries=new Set([new URL('./',scope).href,new URL('index.html',scope).href]);
self.addEventListener('install',event=>event.waitUntil((async()=>{const cache=await caches.open(CACHE);await Promise.all(SHELL.map(async path=>{const url=new URL(path,scope);const response=await fetch(url,{cache:'no-store'});if(!response.ok)throw Error('Precache failed: '+path);await cache.put(canonical(url),response);}));await self.skipWaiting();})()));
self.addEventListener('activate',event=>event.waitUntil((async()=>{const keys=await caches.keys();await Promise.all(keys.filter(k=>k.startsWith('jnr-shell-')&&k!==CACHE).map(k=>caches.delete(k)));await self.clients.claim();})()));
self.addEventListener('message',event=>{if(event.data==='SKIP_WAITING')self.skipWaiting();});
self.addEventListener('fetch',event=>{
 const request=event.request,url=new URL(request.url),key=canonical(url);
 if(request.method!=='GET'||url.origin!==self.location.origin)return;
 // APKs and unknown pages retain their real response, including errors.
 if(!allowed.has(key))return;
 if(request.mode==='navigate'&&!entries.has(key))return;
 event.respondWith((async()=>{try{const response=await fetch(request,{cache:'no-store'});if(response.ok){const copy=response.clone();event.waitUntil(caches.open(CACHE).then(c=>c.put(key,copy)));}return response;}catch(error){const cached=await caches.match(key);if(cached)return cached;if(entries.has(key))return await caches.match(new URL('index.html',scope).href)||await caches.match(new URL('offline.html',scope).href);throw error;}})());
});
